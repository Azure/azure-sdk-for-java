import com.azure.resourcemanager.datatransfer.models.AntivirusRuleset;
import com.azure.resourcemanager.datatransfer.models.AntivirusSolutions;
import com.azure.resourcemanager.datatransfer.models.ArchiveRuleset;
import com.azure.resourcemanager.datatransfer.models.Casing;
import com.azure.resourcemanager.datatransfer.models.DataClassType;
import com.azure.resourcemanager.datatransfer.models.DataSizeRuleset;
import com.azure.resourcemanager.datatransfer.models.FilterType;
import com.azure.resourcemanager.datatransfer.models.FlowProfileProperties;
import com.azure.resourcemanager.datatransfer.models.FlowProfileRulesets;
import com.azure.resourcemanager.datatransfer.models.FlowProfileStatus;
import com.azure.resourcemanager.datatransfer.models.MatchType;
import com.azure.resourcemanager.datatransfer.models.MimeFilterRuleset;
import com.azure.resourcemanager.datatransfer.models.MimeTypeFilter;
import com.azure.resourcemanager.datatransfer.models.TextMatchingRuleset;
import com.azure.resourcemanager.datatransfer.models.TextMatchingRulesetDeny;
import com.azure.resourcemanager.datatransfer.models.XmlFilterRuleset;
import com.azure.resourcemanager.datatransfer.models.XmlReferenceType;
import java.util.Arrays;

/**
 * Samples for FlowProfiles CreateOrUpdate.
 */
public final class FlowProfilesCreateOrUpdateSamples {
    /*
     * x-ms-original-file: 2026-02-06-preview/createOrUpdateFlowProfile.json
     */
    /**
     * Sample code: Creates or updates the FlowPRofile resource.
     * 
     * @param manager Entry point to DataTransferManager.
     */
    public static void
        createsOrUpdatesTheFlowPRofileResource(com.azure.resourcemanager.datatransfer.DataTransferManager manager) {
        manager.flowProfiles()
            .define("testFlowProfile")
            .withRegion("East US")
            .withExistingPipeline("testRG", "testPipeline")
            .withProperties(new FlowProfileProperties().withReplicationScenario(DataClassType.FILES)
                .withStatus(FlowProfileStatus.ENABLED)
                .withDescription("Hello world description")
                .withRulesets(new FlowProfileRulesets()
                    .withAntivirus(new AntivirusRuleset()
                        .withAvSolutions(Arrays.asList(AntivirusSolutions.DEFENDER, AntivirusSolutions.CLAM_AV)))
                    .withArchives(new ArchiveRuleset().withMinimumSizeForExpansion(1L)
                        .withMaximumExpansionSizeLimit(123456L)
                        .withMaximumDepthLimit(12L)
                        .withMaximumCompressionRatioLimit(123.0D))
                    .withDataSize(new DataSizeRuleset().withMaximum(123456789L).withMinimum(1L))
                    .withMimeFilters(new MimeFilterRuleset().withType(FilterType.ALLOW)
                        .withFilters(Arrays.asList(new MimeTypeFilter().withMedia("application/json")
                            .withExtensions(Arrays.asList(".json", ".app", ".js")))))
                    .withXmlFilters(new XmlFilterRuleset().withDefaultNamespace("testnamespace")
                        .withReference(XmlReferenceType.INLINE)
                        .withSchema(
                            "<?xml version=\"1.0\" encoding=\"UTF-8\" ?> <xs:schema xmlns:xs=\"http://www.w3.org/2001/XMLSchema\"></xs:schema>"))
                    .withTextMatching(new TextMatchingRuleset().withDeny(Arrays.asList(
                        new TextMatchingRulesetDeny().withText("hello world")
                            .withMatchType(MatchType.PARTIAL)
                            .withCaseSensitivity(Casing.INSENSITIVE),
                        new TextMatchingRulesetDeny().withText("hello")
                            .withMatchType(MatchType.COMPLETE)
                            .withCaseSensitivity(Casing.SENSITIVE))))))
            .create();
    }
}
